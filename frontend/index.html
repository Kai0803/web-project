class MyStrategy(Strategy):
    params = {"fast": 10, "slow": 30}        # 可調參數（回測/機器人可覆寫）

    def init(self, data):                     # 預先計算指標，data.open/high/low/close/volume/ts 為 list
        self.fast = ta.sma(data.close, self.p.fast)
        self.slow = ta.sma(data.close, self.p.slow)

    def next(self, i, data):                  # 第 i 根 K 線收盤後呼叫
        if ta.crossover(self.fast, self.slow, i):
            return 1.0                        # 目標曝險：1=全倉做多、0=空手、-1=全倉做空（僅永續）
        if ta.crossunder(self.fast, self.slow, i):
            return 0.0
        return None                           # None = 維持現有倉位

可用指標: ta.sma ema rsi macd bollinger atr highest lowest stdev crossover crossunder
可用狀態: self.position（目前曝險 -1~1）、self.entry_price（持倉均價）
訊號在收盤產生、下一根開盤成交（回測），避免未來函數。
